Workshop announcement bar for clearerthinking.org

Generated 2026-09-16 10:16am. Tested by injecting the snippet into the live site in a headless browser.

How it looks

Behaviour

  • Not sticky: it scrolls away with the page. The Wix header keeps its own sticky behaviour.
  • Shows on every page (inserted before Wix's #SITE_CONTAINER), once per page load, also on client-side navigation.
  • The x hides it for 7 days on that browser (localStorage). It stops appearing by itself from the workshop start, 2026-09-30 16:00 UTC.
  • Both the title and the button link to the sign-up page with ?src=sitebar, so these sign-ups are attributed in GuidedTrack's run data.
  • No dependencies, no external requests, keyboard accessible (focus rings, labelled close button).

Install in Wix

  1. In the Wix dashboard open Settings → Advanced → Custom Code (the site may need to be on a plan that allows custom code; it already runs custom code today).
  2. Click + Add Custom Code and paste the whole snippet below. Name it "Workshop announcement bar".
  3. Add code to pages: All pages, Load code once.
  4. Place code in: Body - end.
  5. Apply, then publish the site. Check the homepage in a private window; press the x to make sure it dismisses.

To change the wording, date, link or expiry later, edit the CONFIG block at the top of the script and save the code snippet again in Wix. To take the bar down early, delete the snippet in Custom Code.

<script>
(function () {
  var CONFIG = {
    eyebrow: "Free live workshop",
    title: "Thinking Clearly About Career Change",
    detail: "Wed, Sep 30 · 12 pm ET · on Zoom",
    cta: "Save my spot",
    href: "https://programs.clearerthinking.org/career-workshop/?src=sitebar",
    hideAfter: "2026-09-30T16:00:00Z",   // workshop start (UTC): the bar stops showing from then on
    dismissDays: 7,                       // how long "x" keeps it hidden on this browser
    storageKey: "ctWorkshopBarDismissedAt"
  };
  var ID = "ct-workshop-bar";

  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]; }); }

  function shouldShow() {
    if (Date.now() >= Date.parse(CONFIG.hideAfter)) return false;
    try {
      var dismissedAt = Number(localStorage.getItem(CONFIG.storageKey));
      if (dismissedAt && Date.now() - dismissedAt < CONFIG.dismissDays * 864e5) return false;
    } catch (e) {}
    return true;
  }

  function build() {
    if (document.getElementById(ID) || !shouldShow()) return;
    var bar = document.createElement("div");
    bar.id = ID;
    bar.setAttribute("role", "region");
    bar.setAttribute("aria-label", "Workshop announcement");
    bar.innerHTML =
      '<div class="ctwb-inner">' +
        '<div class="ctwb-text">' +
          '<span class="ctwb-eyebrow">' + esc(CONFIG.eyebrow) + '</span>' +
          '<a class="ctwb-title" href="' + esc(CONFIG.href) + '">' + esc(CONFIG.title) + '</a>' +
          '<span class="ctwb-detail">' + esc(CONFIG.detail) + '</span>' +
        '</div>' +
        '<a class="ctwb-cta" href="' + esc(CONFIG.href) + '">' + esc(CONFIG.cta) + '</a>' +
        '<button class="ctwb-close" type="button" aria-label="Dismiss this announcement">' +
          '<svg viewBox="0 0 20 20" aria-hidden="true"><path d="M5 5l10 10M15 5L5 15" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round"/></svg>' +
        '</button>' +
      '</div>';
    var site = document.getElementById("SITE_CONTAINER");
    if (site && site.parentNode) site.parentNode.insertBefore(bar, site);
    else document.body.insertBefore(bar, document.body.firstChild);
    bar.querySelector(".ctwb-close").addEventListener("click", function () {
      bar.parentNode.removeChild(bar);
      try { localStorage.setItem(CONFIG.storageKey, String(Date.now())); } catch (e) {}
    });
  }

  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", build);
  else build();
})();
</script>
